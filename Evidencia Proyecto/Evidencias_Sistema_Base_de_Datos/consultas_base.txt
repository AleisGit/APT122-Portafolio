== Siniestralidad de la cartera por trimestre
select t.trimestre || 'T ' || t.anio as periodo, sum(t.siniestros_incurridos) as siniestros, sum(t.prima_devengada) as prima, round(sum(t.siniestros_incurridos)::numeric / sum(t.prima_devengada), 4) as siniestralidad from periodo_poliza t group by t.anio, t.trimestre order by t.anio, t.trimestre;
